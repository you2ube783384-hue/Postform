   125→   125→  function validate(): boolean {

   129→   129→    else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email.trim())) e.email = "INVALID EMAIL";
   130→   130→    if (!form.phone.trim()) e.phone = "REQUIRED";
   131→   131→    if (!form.country) e.country = "REQUIRED";
   132→   132→    if (!form.line1.trim()) e.line1 = "REQUIRED";
   133→   133→    if (!form.city.trim()) e.city = "REQUIRED";
   134→   134→    if (!form.postalCode.trim()) e.postalCode = "REQUIRED";
   135→   135→    setErrors(e);
   136→   136→    if (Object.keys(e).length > 0) {
   137→   137→      toast.error("CHECK THE FORM", { description: "Some required fields are missing or invalid." });
   138→   138→      return false;
   139→   139→    }

   174→   174→    const order: OrderData = {

   176→   176→      date: formatOrderDate(),

   178→   178→        name: form.name.trim(),
   179→   179→        email: form.email.trim(),
   180→   180→        phone: form.phone.trim(),
   181→   181→      },
   182→   182→      address,
   183→   183→      items,
   184→   184→      subtotal,
   185→   185→      shipping,

   187→   187→      paymentMethod: form.paymentMethod === "paypal" ? "PayPal" : "Visa / Prepaid Card",
   188→   188→      note: form.note,

   191→   191→    // Save to local profile (optional convenience)

   200→   200→    setConfirmation(order);
   201→   201→    setMailtoUrl(email.mailtoUrl);

   213→   213→      description: "Your email app should have opened — press Send to complete the order request.",
   214→   214→    });
